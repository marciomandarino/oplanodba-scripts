== 🎯 Acessando o Ambiente NON-CDB (ORCL)

▶️ Conectando no banco
$ sqlplus / as sysdba


=================================================================
== 🎯 Criando Usuário em NON-CDB (ORCL)
=================================================================
SQL> create user app1 identified by oracle
    default tablespace users
    temporary tablespace temp;

▶️ Validando criação

SQL> set lines 200
SQL> set pages 200
SQL> col username format a20
SQL> col account_status format a20

SQL> select username as username,
           account_status as account_status
    from dba_users
    where username = 'APP1';


=================================================================
== 🎯 Teste de Acesso (Erro Esperado)
=================================================================
▶️ Saindo do SQL*Plus

SQL> exit

▶️ Tentando conectar com usuário recém criado

$ sqlplus app1/oracle

▶️ Resultado:

ORA-01045: user APP1 lacks CREATE SESSION privilege; logon denied


=================================================================
== 🎯 Concedendo Privilégio Mínimo
=================================================================
▶️ Voltando como SYS

$ . oraenv <<< ORCL
$ sqlplus / as sysdba

SQL> grant create session to app1;

▶️ Testando novamente

SQL> exit

$ sqlplus app1/oracle


💡 Nesta série vamos aprofundar privilégios mais à frente


=================================================================
== 🎯 Usuário Comum (Todos os PDBs)
=================================================================
▶️ Ajustando ambiente para CDB

$ . oraenv <<< CDB1
$ sqlplus / as sysdba

▶️ Criando usuário comum

SQL> create user c##app identified by oracle container = all;
SQL> grant create session to c##app container = all;
SQL> grant set container to c##app container = all;

▶️ Validando

SQL> set lines 200
SQL> set pages 200
SQL> col username format a20
SQL> col common format a6

SQL> select username as username,
           common as common
    from cdb_users
    where username like 'C##APP';


=================================================================
== 🎯 Usuários em Alguns PDBs (Local User)
=================================================================
▶️ Usuários locais em PDBs específicos

SQL> alter session set container = HUMAN_RESOURCES;

SQL> create user app_local identified by oracle;

SQL> alter session set container = CUSTOMER_ORDERS;

SQL> create user app_local identified by oracle;

▶️ Mesmo nome, usuários diferentes
▶️ Cada PDB possui seu próprio usuário

== 🎯 Scripts de Consulta (users.sql)
=================================================================
/* Script: users.sql | Autor: Marcio Mandarino
 Descrição: Exibe os usuários do banco filtrando pelo nome informado
 Parâmetros: 1   (&1 = parte do nome do usuário)
 Exemplo de uso: @users app
 Observações: Utiliza LIKE para busca parcial (case insensitive)
 ---------------------------------------------------------------------
 Histórico:
 --------------------------------------------------------------------- */

SET LINES 900
SET VERIFY OFF

col username              form a30
col account_status        form a20
col default_tablespace    form a20
col temporary_tablespace  form a20
col last_login            form a22
col profile               form a30
col oracle_maintained     form a18

SELECT username,
       created,
       profile,
       to_char(last_login,'DD/MM/RRRR HH24:MI:SS') last_login,
       oracle_maintained,
       common,
       account_status,
       lock_date,
       default_tablespace,
       temporary_tablespace
FROM dba_users
WHERE username LIKE UPPER('%&1%')
ORDER BY username;


=================================================================
== 🎯 Scripts de Consulta (users2.sql)
=================================================================
/* Script: users2.sql | Autor: Marcio Mandarino
 Descrição: Exibe os usuários do banco com filtro dinâmico e ordenação definida pelo usuário
 Parâmetros: 2   (&1 = filtro WHERE | &2 = ORDER BY)
 Exemplo de uso: @users2 "oracle_maintained='N'" "username"
 Observações: Utiliza a visão DBA_USERS para consulta detalhada
 ---------------------------------------------------------------------
 Histórico:
 --------------------------------------------------------------------- */

SET LINES 900
SET VERIFY OFF

col username              form a30
col account_status        form a20
col default_tablespace    form a20
col temporary_tablespace  form a20
col last_login            form a22
col profile               form a30
col oracle_maintained     form a18

SELECT username,
       created,
       profile,
       to_char(last_login,'DD/MM/RRRR HH24:MI:SS') last_login,
       oracle_maintained,
       common,
       account_status,
       lock_date,
       default_tablespace,
       temporary_tablespace
FROM dba_users
WHERE &1
ORDER BY &2;
