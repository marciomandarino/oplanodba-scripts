=================================================================
== 🎯 Visão Geral
=================================================================
▶️ O Database Link permite consultar dados de outro banco Oracle direto de uma query SQL — sem precisar exportar ou copiar dados
▶️ Útil quando você precisa integrar informações entre PDBs ou bancos diferentes em tempo real


=================================================================
== 🎯 Setup
=================================================================
-- Vamos criar um link da PDB HUMAN_RESOURCES (CDB1) para a PDB SALES_HISTORY (CDB2)

$ . oraenv <<< CDB1
$ sqlplus / as sysdba

SQL> ALTER SESSION SET CONTAINER = HUMAN_RESOURCES;


=================================================================
== 🎯 Execução
=================================================================
-- Cria o database link apontando para o usuário sh na PDB SALES_HISTORY
SQL> CREATE DATABASE LINK to_sales_history
     CONNECT TO sh IDENTIFIED BY oracle
     USING 'localhost:1522/SALES_HISTORY';


=================================================================
== 🎯 Validação
=================================================================
-- Testar a conexão acessando dados pelo link
SQL> set lines 200
SQL> set pages 200
SQL> col owner         format a20
SQL> col segment_name  format a30
SQL> col segment_type  format a15

SQL> SELECT owner,
            segment_name,
            segment_type
     FROM   dba_segments@to_sales_history
     WHERE  owner = 'SH'
     AND    ROWNUM <= 5;

-- Confirmar o link no catálogo
SQL> col db_link    format a30
SQL> col username   format a20
SQL> col host       format a40

SQL> SELECT db_link,
            username,
            host
     FROM   dba_db_links;
